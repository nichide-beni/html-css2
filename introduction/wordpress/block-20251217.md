> functions.php

```php
<?php

/**
 * 魚の種類 (fish_name) のカスタムフィールドの値のみを返すショートコード
 *
 * 使用例: [fish_name_value]
 *
 * @return string 魚の種類の値、または空文字列
 */
function get_fish_name_value_shortcode() {
  global $post; 
  $post_id = $post ? $post->ID : get_the_ID();

  if ( get_post_type($post_id) !== 'fish' || !$post_id ) {
    return '';
  }

  // 標準のカスタムフィールド (post_meta) から値を取得 (第三引数 true で単一の値を取得)
  $fish_name = get_post_meta($post_id, 'fish_name', true);

  return !empty($fish_name) ? esc_html($fish_name) : '';
}
add_shortcode('fish_name_value', 'get_fish_name_value_shortcode');

/**
 * 魚の大きさ (fish_size) のカスタムフィールドの値のみを返すショートコード
 *
 * 使用例: [fish_size_value]
 *
 * @return string 魚の大きさの値、または空文字列
 */
function get_fish_size_value_shortcode() {
  global $post; 
  $post_id = $post ? $post->ID : get_the_ID();
    
  if ( get_post_type($post_id) !== 'fish' || !$post_id ) {
    return '';
  }

  $fish_size = get_post_meta($post_id, 'fish_size', true);

  // 値があればそれを、なければ空文字列を返す
  return !empty($fish_size) ? esc_html($fish_size) : '';
}
add_shortcode('fish_size_value', 'get_fish_size_value_shortcode');
```
